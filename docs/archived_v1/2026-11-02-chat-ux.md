**Tech Stack:** Kotlin, Swing, IntelliJ Platform (`BasePlatformTestCase`, `AsyncProcessIcon`, `JBColor`), Gradle, Kover.

**Spec:** the user request (pasted 2026-11-02): 5 items — user bubble alignment, busy state, active worker role, no repeated tool output, Up-arrow recall.

private val userBubbleBg = JBColor(Color(0xDCE9FF), Color(0x2F4468))

private fun addMessage(role: String, content: String): JTextArea {
    val isUser = role == "user"
    val area = JTextArea(if (isUser) content else "$role: $content")
    area.isEditable = false; area.lineWrap = true; area.wrapStyleWord = true
    area.font = UIUtil.getLabelFont()
    area.background = if (isUser) userBubbleBg else UIUtil.getPanelBackground()
    area.border = JBUI.Borders.empty(4, 8)
    val row = object : JPanel(BorderLayout()) {
        override fun getMaximumSize() = Dimension(Int.MAX_VALUE, preferredSize.height)
    }
    row.isOpaque = false
    row.alignmentX = LEFT_ALIGNMENT
    if (isUser) {
        row.border = JBUI.Borders.emptyLeft(60)       // keeps bubble off the left edge
        row.add(area, BorderLayout.EAST)
    } else row.add(area, BorderLayout.CENTER)
    messagesPanel.add(row)
    refreshMessages()
    return area
}

internal fun messageAreas(): List<JTextArea> =
    messagesPanel.components.flatMap { c ->
        if (c is JTextArea) listOf(c) else (c as? JPanel)?.components?.filterIsInstance<JTextArea>().orEmpty()
    }
```
Replace the leading `emptyLeft(60)` with a width-capping `getPreferredSize` if text wrapping looks wrong in `runIde` (see Task 6 manual check).

package com.adb.stewart.chat
import com.adb.stewart.agents.AgentRole

object ActivityFormat {
    private const val MAX = 40
    fun label(role: AgentRole?, context: String?, tool: String?): String {
        if (role == null) return "Working…"
        val name = role.name.lowercase().replaceFirstChar { it.uppercase() }
        val ctx = context?.substringBefore(" → ")?.substringBefore(":")?.trim()?.take(MAX)?.ifEmpty { null }
        // tool name wins over context while a tool runs
        val detail = tool ?: ctx
        return if (detail == null) name else "$name · $detail"
    }
}
```

```kotlin
class PromptHistory {
    private var entries = emptyList<String>(); private var idx = -1
    val browsing get() = idx >= 0
    fun reset(e: List<String>) { entries = e; idx = -1 }
    fun older(): String? {
        if (entries.isEmpty()) return null
        val next = if (idx < 0) entries.lastIndex else idx - 1
        if (next < 0) return null
        idx = next; return entries[idx]
    }
    fun newer(): String? {
        if (idx < 0) return null
        if (idx == entries.lastIndex) { idx = -1; return "" }
        idx++; return entries[idx]
    }
    fun stopBrowsing() { idx = -1 }
}
```
